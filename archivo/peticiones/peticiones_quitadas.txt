/* ---- estado y errores de peticiones ---- */
let misPeticiones = [];
let peticionesPendientes = [];
function nombreTipoPeticion(tipo, cell){ return tipo === 1 ? 'Partida' : ((cell && TERMINADO_LABEL[cell.area]) || 'Terminado'); }
const ERRORES_PETICION = {
  ocupada: 'Ese departamento ya tiene a otra persona, o alguien más ya lo pidió primero.',
  ya_enviada: 'Esa petición ya la enviaste, queda esperando respuesta.',
  invalida: 'Eso no se puede pedir con el estado actual del departamento. Recarga la página por si cambió.',
  sin_sesion: 'Tu sesión venció. Vuelve a iniciar sesión.'
};


/* ---- botón de petición ---- */
function tienePeticionMia(celdaId, tipo){ return misPeticiones.some(p => p.celda === celdaId && p.tipo === tipo); }
function peticionBtnHtml(cell, st){
  if(!PETICIONES_ACTIVAS || !workerSession) return '';
  const esMia = st.trabajador === workerSession, libre = !st.trabajador;
  if(!esMia && !libre) return '';
  const estado = resolveEstado(st), frg = esFrague(cell);
  let tipo = null;
  if(estado === 'pendiente') tipo = frg ? 2 : 1;
  else if(estado === 'proceso' && esMia) tipo = 2;
  if(!tipo) return '';
  const nombre = nombreTipoPeticion(tipo, cell);
  if(tienePeticionMia(cell.id, tipo)) return `<div class="peticion-estado"><i class="ic ic-reloj"></i> Petición de ${nombre} enviada</div>`;
  return `<button type="button" class="btn small peticion-btn" data-peticion="${cell.id}" data-tipo="${tipo}">Solicitar ${nombre}</button>`;
}


/* ---- funciones de peticiones ---- */
async function cargarMisPeticiones(){
  if(!PETICIONES_ACTIVAS || !workerSession || !DB_URL || !sessionToken) return;
  try{
    const j = await llamarDb('misPeticiones');
    if(j && j.ok){
      const antes = misPeticiones.length;
      misPeticiones = j.peticiones || [];

      if(misPeticiones.length < antes) await refrescarDatosTrabajador(); else render();
    }
  }catch(e){}
}
async function enviarPeticion(celdaId, tipo){
  { const mot = !getState(celdaId).partida ? motivoBloqueoPartida(cellById[celdaId], tipo === 1 || (esFrague(cellById[celdaId]) && !getState(celdaId).trabajador)) : null; if(mot){ avisoToast(textoBloqueoPartida(mot, cellById[celdaId]), 'error'); return; } }
  if(!DB_URL){ avisoToast('Hace falta la base de datos conectada para enviar peticiones.'); return; }
  mostrarCargando('Enviando petición…');
  try{
    const j = await llamarDb('crearPeticion', {celda: celdaId, tipo});
    if(j && j.ok){ misPeticiones.push({celda: celdaId, tipo}); avisoToast('Petición enviada. Queda esperando que Admin la revise.'); }
    else avisoToast(ERRORES_PETICION[j && j.error] || 'No se pudo enviar la petición. Prueba de nuevo.');
  }catch(e){ avisoToast('No se pudo enviar (revisa la conexión).'); }
  finally{ ocultarCargando(); render(); }
}
function actualizarBadgePeticiones(){
  const b = document.getElementById('peticionesBadge'); if(!b) return;
  b.textContent = peticionesPendientes.length;
  b.style.display = peticionesPendientes.length ? 'inline-flex' : 'none'; // @@molde-capsulas
}
async function cargarPeticionesAdmin(){
  if(!PETICIONES_ACTIVAS || !isAdmin || !DB_URL || !sessionToken) return;
  try{
    const j = await llamarDb('listarPeticiones');
    if(j && j.ok){
      peticionesPendientes = j.peticiones || []; actualizarBadgePeticiones();
      if(document.getElementById('peticionesOverlay').classList.contains('show')) renderPeticiones();
    }
  }catch(e){}
}
let petTab = 'pend';
let historialPeticiones = [];
async function cargarHistorialPeticiones(){
  if(!isAdmin || !DB_URL || !sessionToken) return;
  try{
    const j = await llamarDb('historialPeticiones');
    if(j && j.ok){ historialPeticiones = j.peticiones || []; if(petTab === 'hist') renderPeticiones(); }
  }catch(e){}
}
function cambiarTabPeticiones(tab){
  petTab = tab;
  document.getElementById('petTabPend').classList.toggle('primary', tab === 'pend');
  document.getElementById('petTabHist').classList.toggle('primary', tab === 'hist');
  document.getElementById('petAyuda').style.display = tab === 'pend' ? '' : 'none';
  renderPeticiones();
  if(tab === 'hist') cargarHistorialPeticiones();
}
function renderHistorialPeticiones(el){
  if(!historialPeticiones.length){
    el.innerHTML = '<div style="border:1px dashed var(--border-strong);padding:24px;text-align:center;color:var(--sub);">Todavía no hay peticiones resueltas.</div>';
    return;
  }
  el.innerHTML = historialPeticiones.map(p => {
    const cell = cellById[p.celda];
    const donde = cell ? `${liqLabelForCell(cell)} — ${AREA_LABEL[cell.area]||cell.area} (${cell.piso}, ${cell.categoria})` : p.celda;
    const ok = p.estado === 1;
    return `<div class="peticion-item">
      <div>${ok ? '<i class="ic ic-ok"></i> Aceptada' : '<i class="ic ic-no"></i> Rechazada'} · <b>${p.trabajador}</b> pidió <b>${nombreTipoPeticion(p.tipo, cell)}</b></div>
      <div class="sub">${donde}</div>
      <div class="sub">Pedida: ${p.fecha ? fechaCortaHora(p.fecha) : '—'} · Resuelta: ${p.resuelta ? fechaCortaHora(p.resuelta) : '—'}</div>
    </div>`;
  }).join('');
}
function renderPeticiones(){
  const el = document.getElementById('peticionesLista');
  if(petTab === 'hist'){ renderHistorialPeticiones(el); return; }
  if(!peticionesPendientes.length){
    el.innerHTML = '<div style="border:1px dashed var(--border-strong);padding:24px;text-align:center;color:var(--sub);">No hay peticiones esperando.</div>';
    return;
  }
  el.innerHTML = peticionesPendientes.map(p => {
    const cell = cellById[p.celda];
    const donde = cell ? `${liqLabelForCell(cell)} — ${AREA_LABEL[cell.area]||cell.area} (${cell.piso}, ${cell.categoria})` : p.celda;
    return `<div class="peticion-item">
      <div><b>${p.trabajador}</b> pide <b>${nombreTipoPeticion(p.tipo, cell)}</b></div>
      <div class="sub">${donde}${p.fecha ? ' · ' + fechaCortaHora(p.fecha) : ''}</div>
      <div class="acciones"><button class="btn small primary" data-pet-aceptar="${p.id}">Aceptar</button><button class="btn small" data-pet-rechazar="${p.id}">Rechazar</button></div>
    </div>`;
  }).join('');
  el.querySelectorAll('[data-pet-aceptar]').forEach(b => b.addEventListener('click', () => resolverPeticionUI(b.getAttribute('data-pet-aceptar'), true)));
  el.querySelectorAll('[data-pet-rechazar]').forEach(b => b.addEventListener('click', () => resolverPeticionUI(b.getAttribute('data-pet-rechazar'), false)));
}
async function resolverPeticionUI(id, aceptar){
  const p = peticionesPendientes.find(x => x.id === id); if(!p || guardandoAhora) return;
  mostrarCargando('Guardando…');
  try{
    if(aceptar){
      const st = getState(p.celda);
      const frgP = esFrague(cellById[p.celda]);
      if((p.tipo === 1 || frgP) && st.trabajador && st.trabajador !== p.trabajador){
        avisoToast('Ese departamento ya tiene a otra persona asignada. Rechaza esta petición.'); return;
      }
      { const mot = !st.partida ? motivoBloqueoPartida(cellById[p.celda], p.tipo === 1 || (frgP && !st.trabajador)) : null; if(mot){ avisoToast(textoBloqueoPartida(mot, cellById[p.celda]) + ' Rechaza esta petición o espera.', 'error'); return; } }
      const ahora = Date.now();
      if(p.tipo === 1){ st.trabajador = p.trabajador; st.partida = true; st.partidaTs = st.partidaTs || ahora; }
      else { if(frgP && !st.trabajador) st.trabajador = p.trabajador; st.partida = true; st.partidaTs = st.partidaTs || ahora; st.terminado = true; st.terminadoTs = ahora; matCheckTerminado(p.celda); }
      markCellDirty(p.celda);
      const r = await enqueueSave();
      if(DB_URL && !r.dbOk){ avisoToast('No se pudo guardar el cambio; la petición sigue pendiente.'); return; }
    }
    const j = await llamarDb('resolverPeticion', {id, estado: aceptar ? 1 : 2});
    if(j && (j.ok || j.error === 'ya_resuelta')){
      peticionesPendientes = peticionesPendientes.filter(x => x.id !== id);

      if(aceptar && (p.tipo === 1 || frgP)){
        const otras = peticionesPendientes.filter(x => x.celda === p.celda && x.tipo === (p.tipo));
        for(const o of otras){ await llamarDb('resolverPeticion', {id: o.id, estado: 2}); }
        peticionesPendientes = peticionesPendientes.filter(x => !(x.celda === p.celda && x.tipo === p.tipo));
      }
      actualizarBadgePeticiones(); renderPeticiones(); render();
      historialPeticiones = [];
    } else avisoToast('No se pudo marcar la petición como resuelta. Prueba de nuevo.');
  }catch(e){ avisoToast('No se pudo completar (revisa la conexión).'); }
  finally{ ocultarCargando(); }
}
