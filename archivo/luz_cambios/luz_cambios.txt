// @@cambio-ajeno: trae lo guardado por otra cuenta y cambia solo las celdas distintas (por ID), sin redibujar la lista ni recargar.
// No toca celdas con cambios sin guardar ni los ajustes; si algo no se puede leer, no hace nada.
async function aplicarCambiosAjenos(){
  if(!DB_URL || !sessionToken) return -1;
  const remote = await dbLoad();
  if(!remote || !remote.items) return -1;
  const ids = [];
  Object.keys(remote.items).forEach(id => {
    if(!cellById[id] || dirtyCellIds.has(id) || colaTieneCelda(id)) return;
    if(JSON.stringify(state[id] || null) === JSON.stringify(remote.items[id])) return;
    state[id] = remote.items[id]; baseRec[id] = recordForCell(cellById[id]); ids.push(id);
  });
  if(ids.length) refrescarCeldas(ids);
  return ids.length;
}
// @@luz-cambios: luz de prueba. Pregunta cada 30 s (solo con la pestaña a la vista) cuál fue el último cambio guardado.
// No recarga datos ni toca los otros temporizadores: solo avisa. Al tocar el aviso ámbar, recarga la página.
(function(){
  const el = document.getElementById('luzCambios'); if(!el) return;
  const msg = el.querySelector('.lzm');
  let base = null, hayOtro = false, ocupado = false, ultimo = 0;
  const quienCorto = q => { const t = String(q || '').split(' - '); return t.length > 1 ? nombreCorto(t.slice(1).join(' - ')) : (t[0] || 'Alguien'); };
  const poner = (est, txt, tip) => { el.className = est; el.hidden = false; msg.textContent = txt || ''; el.title = tip || ''; };
  async function mirar(){
    if(!DB_URL || !sessionToken || ocupado) return;
    ocupado = true; ultimo = Date.now();
    try{
      const r = await llamarDb('ultimoCambio', {});
      if(!r || !r.ok){ if(!hayOtro) poner('gris', '', 'Sin conexión con el aviso de cambios'); return; }
      const firma = (r.ts || 0) + '|' + (r.n || 0);
      if(base === null){ base = firma; }
      else if(firma !== base){
        base = firma;
        if(!r.mio){
          hayOtro = true;
          const lugar = r.celda ? ' · ' + String(r.celda).split(' · ').slice(0, 2).join(' · ') : '';
          const quien = quienCorto(r.quien) + ' hizo un cambio' + lugar;
          poner('ambar', quien, 'Actualizando…');
          aplicarCambiosAjenos().then(n => {   // @@cambio-ajeno
            if(n < 0) poner('ambar', quien, 'No se pudo actualizar solo: toca para recargar la página');
            else{ hayOtro = false; poner('ambar', quien + (n ? ' · ya está en pantalla' : ''), 'Cambio aplicado sin recargar'); setTimeout(() => { if(!hayOtro) poner('verde', '', 'Al día'); }, 6000); }
            if(n < 0) hayOtro = true;
          }).catch(() => { hayOtro = true; poner('ambar', quien, 'No se pudo actualizar solo: toca para recargar la página'); });
        }
      }
      if(!hayOtro) poner('verde', '', 'Al día: nadie más ha cambiado nada desde que abriste');
    }catch(e){ if(!hayOtro) poner('gris', '', 'Sin conexión con el aviso de cambios'); }
    finally{ ocupado = false; }
  }
  el.addEventListener('click', () => { if(el.classList.contains('ambar') && hayOtro) location.reload(); });
  setInterval(() => { if(!document.hidden) mirar(); }, 30000);
  document.addEventListener('visibilitychange', () => { if(!document.hidden && Date.now() - ultimo > 30000) mirar(); });
  setTimeout(mirar, 4000);
})();

function colaTieneCelda(id){
  try{ return colaLeer().some(e => e && e.cuerpo && Array.isArray(e.cuerpo.records) && e.cuerpo.records.some(r => r && r.id === id)); }catch(e){ return true; }
}
