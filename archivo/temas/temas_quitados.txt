function dzAreasHtml(fb, d, filtering){
  return (d.esPasillo ? ['Pasillos'] : areasVista(fb)).map(area => {
    const c = d.cells[area];
    let cuerpo;
    if(!c) cuerpo = '<span class="cell-empty">—</span>';
    else if(filtering && !passesFilters(cellById[c.id])) cuerpo = '<span class="cell-empty">(no coincide)</span>';
    else{
      const cell = cellById[c.id], st = getState(c.id);
      cuerpo = vistaVentana() ? dzFila(cell, st) : `<div class="cell-td ${claseEstadoTd(st)}">${cellHtml(cell)}</div>`;
    }
    return `<div class="dz-fl"><span class="dz-ar">${AREA_LABEL[area] || area}</span><div class="dz-cl">${cuerpo}</div></div>`;
  }).join('');
}
function dzSegHtml(fb, d, clase){
  if(d.esPasillo) return `<span class="${clase}"><i class="st-${resolveEstado(getState(d.cells.Pasillos.id))}"></i></span>`;
  return `<span class="${clase}">${areasVista(fb).map(area => { const c = d.cells[area]; return c ? `<i class="st-${resolveEstado(getState(c.id))}"></i>` : '<i></i>'; }).join('')}</span>`;
}
function dzNum(fb, d){
  return `<span class="nf">${d.dpto}</span><span class="ns"><i${fb.pisoNum === 25 ? ' class="az"' : ''}>${fb.pisoNum === 25 ? '' : esc(String(fb.piso).replace(/^Piso\s*/i, ''))}</i>${esc(dptoCortoPanel(fb.piso, d.dpto))}</span>`;
}
function dzMatClase(fb, d){
  if(!isAdmin || fb.categoria !== 'Cerámica') return '';
  const u = matUnidadPorClave()[fb.piso + '|' + d.dpto], nv = u ? matNivel(u) : null;
  return nv ? (nv === 'rojo' ? ' dz-m-rojo' : ' dz-m-amar') : '';
}
function dzMatHtml(fb, d){
  if(!isAdmin || fb.categoria !== 'Cerámica') return '';
  const key = fb.piso + '|' + d.dpto, u = matUnidadPorClave()[key], nv = u ? matNivel(u) : null;
  return nv ? `<span class="dz-mat ${nv === 'rojo' ? 'rojo' : 'amarillo'}" data-dzmat="${esc(key)}" title="${matTitulo(nv)} · checks de materiales pendientes: ${matChecksPendientes(u)}">${matChecksPendientes(u)}</span>` : '';
}
function dzFiltrados(fb){
  const filtering = matchesActiveFilters();
  return {filtering, deptRows: fb.depts.filter(d => !filtering || areasVista(fb).some(area => { const c = d.cells[area]; return c && passesFilters(cellById[c.id]); }))};
}
function dzTitulo(fb, n){
  return `<div class="floor-title"><h2>${fb.piso}</h2>${pisoFlechasHtml(fb.piso)}<div class="line"></div><div class="badge">${n} departamento(s)</div></div>`;
}
function renderFloorCards(fb){
  const {filtering, deptRows} = dzFiltrados(fb);
  if(!deptRows.length) return '';
  // Igual que en Calidad: 4 tarjetas arriba y abajo las que sobran + el Pasillo al final
  const filas = deptRows.concat((fb.pasillo && cellById[fb.pasillo.id] && (!filtering || passesFilters(cellById[fb.pasillo.id]))) ? [{dpto: 'Pasillo', esPasillo: true, cells: {Pasillos: fb.pasillo}}] : []);
  return `<div class="floor-block" data-piso="${esc(fb.piso)}">${dzTitulo(fb, deptRows.length)}<div class="dz-cards">${filas.map(d => d.esPasillo ? `
    <div class="dz-card"><div class="dz-ch"><b>Pasillo</b>${dzSegHtml(fb, d, 'dz-seg')}</div>${dzAreasHtml(fb, d, filtering)}</div>` : `
    <div class="dz-card"><div class="dz-ch${dzMatClase(fb, d)}"><b>${dzNum(fb, d)}</b>${dzMatHtml(fb, d)}${dzSegHtml(fb, d, 'dz-seg')}</div>${dzAreasHtml(fb, d, filtering)}</div>`).join('')}</div></div>`;
}
function renderFloorMapa(fb){
  const {filtering, deptRows} = dzFiltrados(fb);
  if(!deptRows.length) return '';
  if(!deptRows.some(d => d.dpto === dzSel[fb.piso])) dzSel[fb.piso] = deptRows[0].dpto;
  const sel = deptRows.find(d => d.dpto === dzSel[fb.piso]);
  const leg = ['pendiente', 'proceso', 'entregado', 'pagado'].map(e => `<span><i class="st-${e}"></i>${e === 'proceso' ? 'Partida' : e === 'entregado' ? 'Terminado' : e[0].toUpperCase() + e.slice(1)}</span>`).join('');
  return `<div class="floor-block" data-piso="${esc(fb.piso)}">${dzTitulo(fb, deptRows.length)}<div class="dz-leg">${leg}</div>
    <div class="dz-mapa">${deptRows.map(d => `<button type="button" class="dz-tile${d.dpto === sel.dpto ? ' on' : ''}${dzMatClase(fb, d)}" data-dzpiso="${esc(fb.piso)}" data-dzdpto="${esc(d.dpto)}"><b>${dzNum(fb, d)}</b>${dzSegHtml(fb, d, 'dz-bars')}</button>`).join('')}</div>
    <div class="dz-det"><h3 class="${dzMatClase(fb, sel).trim()}">Departamento ${sel.dpto} ${dzMatHtml(fb, sel)}</h3>${dzAreasHtml(fb, sel, filtering)}</div></div>`;
}
document.getElementById('content').addEventListener('click', (e) => {
  const t = e.target.closest && e.target.closest('.dz-tile');
  if(!t) return;
  const piso = t.getAttribute('data-dzpiso');
  dzSel[piso] = t.getAttribute('data-dzdpto');
  const fb = FLOOR_BLOCKS.find(f => f.categoria === activeCategoria && f.piso === piso), bl = t.closest('.floor-block');
  if(fb && bl) bl.outerHTML = renderFloorMapa(fb);
});

// Marco dorado es un tema solo nocturno: fuerza la noche y, al salir, devuelve el modo que tenías.
let modoAntesMarco = null;
function aplicarNocheMarco(on){
  const b = document.body, h = document.documentElement, btn = document.getElementById('toggleDark');
  if(on){
    if(modoAntesMarco === null) modoAntesMarco = b.classList.contains('dark');
    b.classList.add('dark'); h.classList.add('dark'); btn.innerHTML = '<i class="ic ic-sol"></i>';
  }else if(modoAntesMarco !== null){
    b.classList.toggle('dark', modoAntesMarco); h.classList.toggle('dark', modoAntesMarco); btn.innerHTML = modoAntesMarco ? '<i class="ic ic-sol"></i>' : '<i class="ic ic-luna"></i>';
    modoAntesMarco = null;
  }
}
// Temas 5 a 8: cada uno cambia la forma de ver pisos y departamentos y funciona de día y de noche.
const TEMAS_DZ = {tema6: 'A', tema7: 'B', tema8: 'C', tema9: 'D'};
let doradoOn = false;
try{ doradoOn = localStorage.getItem('progresoObraDorado') === '1'; }catch(err){}
function aplicarDorado(){
  const cl = document.body.classList, hal = cl.contains('tema5'), btn = document.getElementById('doradoBtn');
  cl.toggle('tema4', doradoOn && !hal);
  btn.innerHTML = '<i class="ic ib ic-estrella"></i>Dorado: ' + (doradoOn && !hal ? 'Sí' : 'No');
  btn.classList.toggle('primary', doradoOn && !hal);
  btn.disabled = hal;
  btn.title = hal ? 'Halloween ya trae sus propios colores' : 'Marco dorado: se puede usar de día y de noche';
  actualizarPrincipalBtn();
}
function temaActual(){
  const v = document.getElementById('temaSelector').value;
  return {tema: v, dorado: !!doradoOn && v !== 'tema5'};
}
function actualizarPrincipalBtn(){
  const b = document.getElementById('principalBtn'); if(!b) return;
  const t = temaPrincipal, c = temaActual();
  const igual = !!t && t.tema === c.tema && !!t.dorado === c.dorado;
  b.innerHTML = '<i class="ic ib ic-estrella"></i>' + (igual ? 'Tema principal' : 'Asignar como tema principal');
  b.classList.toggle('primary', igual);
}
function aplicarTemaPrincipal(){
  if(!temaPrincipal) return;
  let propio = null;
  try{ propio = localStorage.getItem('progresoObraTemaV3'); }catch(err){}
  if(propio === null){
    doradoOn = !!temaPrincipal.dorado;
    const tp = temaPrincipal.tema || 'tema9';
    aplicarTema(tp);
    document.getElementById('temaSelector').value = tp;
  }
  actualizarPrincipalBtn();
}
function aplicarTema(v){
  if(v === 'tema5') cargarTemas(); // @@temas-carga
  const cl = document.body.classList;
  cl.remove('tema1', 'tema2', 'tema3', 'tema5', 'tema6', 'tema7', 'tema8', 'tema9', 'dz-A', 'dz-B', 'dz-C', 'dz-D');
  if(TEMAS_DZ[v]) cl.add('tema3', v, 'dz-' + TEMAS_DZ[v]);
  else{
    if(v) cl.add(v);
    if(v === 'tema5') cl.add('tema3');
  }
  aplicarNocheMarco(v === 'tema5');
  aplicarDorado();
  actualizarPrincipalBtn();
}
